cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(edge_provision LANGUAGES C)
  enable_testing()
endif()
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
add_library(edge_developer_credentials STATIC developer_credentials.c)
target_include_directories(edge_developer_credentials PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(edge_developer_credentials PUBLIC OpenSSL::Crypto)
target_compile_features(edge_developer_credentials PUBLIC c_std_99)
add_executable(edge-provision main.c)
target_link_libraries(edge-provision PRIVATE edge_developer_credentials)
if(WIN32)
  target_link_libraries(edge-provision PRIVATE advapi32)
  find_file(EDGE_CRYPTO_RUNTIME NAMES libcrypto-3-x64.dll
    HINTS "${OPENSSL_ROOT_DIR}/bin" "${OPENSSL_ROOT_DIR}")
  if(NOT EDGE_CRYPTO_RUNTIME)
    message(FATAL_ERROR "OpenSSL runtime DLL required for edge-provision")
  endif()
  add_custom_command(TARGET edge-provision POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EDGE_CRYPTO_RUNTIME}" $<TARGET_FILE_DIR:edge-provision>
    VERBATIM)
endif()
option(EDGE_PROVISION_TESTS "Build native developer credential conversion tests" ON)
if(EDGE_PROVISION_TESTS)
  add_executable(edge-provision-tests tests.c)
  target_link_libraries(edge-provision-tests PRIVATE edge_developer_credentials)
  if(WIN32)
    add_dependencies(edge-provision-tests edge-provision)
  endif()
  add_test(NAME edge-provision-conversion COMMAND edge-provision-tests)
  add_test(NAME edge-provision-cli COMMAND ${CMAKE_COMMAND}
    "-DTOOL=$<TARGET_FILE:edge-provision>" "-DFIXTURE=$<TARGET_FILE:edge-provision-tests>"
    "-DWORK=${CMAKE_CURRENT_BINARY_DIR}/cli-tests" -P "${CMAKE_CURRENT_SOURCE_DIR}/cli-tests.cmake")
endif()
foreach(target edge_developer_credentials edge-provision)
  if(MSVC)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
install(TARGETS edge-provision RUNTIME DESTINATION bin)
