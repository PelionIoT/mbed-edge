cmake_minimum_required(VERSION 3.22)
project(windows_named_pipe_example C)
if (NOT WIN32)
  message(FATAL_ERROR "This example uses native Windows named pipes")
endif ()
add_executable(named-pipe-counter-example named-pipe-counter.c)
target_compile_definitions(named-pipe-counter-example PRIVATE
  _WIN32_WINNT=0x0600 WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(named-pipe-counter-example PRIVATE advapi32)
if (MSVC)
  target_compile_options(named-pipe-counter-example PRIVATE /W4 /WX)
endif ()
enable_testing()
add_test(NAME native-c-named-pipe-counter COMMAND named-pipe-counter-example --self-test)
add_test(NAME native-c-named-pipe-counter-generic-access COMMAND named-pipe-counter-example --self-test-generic)
set_tests_properties(native-c-named-pipe-counter native-c-named-pipe-counter-generic-access PROPERTIES TIMEOUT 25)
