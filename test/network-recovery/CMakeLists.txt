cmake_minimum_required(VERSION 3.22)
project(edge_network_policy_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
add_executable(edge-network-policy-tests policy_tests.c ../../edge-core/edge_network_policy.c)
target_include_directories(edge-network-policy-tests PRIVATE ../../edge-core)
if(MSVC)
  target_compile_options(edge-network-policy-tests PRIVATE /W4 /WX)
else()
  target_compile_options(edge-network-policy-tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME shared-network-recovery-policy COMMAND edge-network-policy-tests)
set_tests_properties(shared-network-recovery-policy PROPERTIES TIMEOUT 10)
